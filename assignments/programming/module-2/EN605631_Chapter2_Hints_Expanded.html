Starter R code:

# Starter: Axioms & finite additivity
# Build a finite probability space {ω_i, p_i}
S <- data.frame(omega = 1:6, p = c(0.1, 0.2, 0.15, 0.25, 0.2, 0.1))  # example

P <- function(A) sum(S$p[S$omega %in% A])  # probability of event A (as a set of outcomes)

# Check axioms
P(1:6)                         # Axiom 2: should be 1
P(integer(0))                  # P(∅) should be 0
A <- c(1,2); B <- c(3,4)       # disjoint events
P(union(A,B)); P(A) + P(B)     # Axiom 3 for disjoint unions

# Complement
Ac <- setdiff(S$omega, A)
P(A) + P(Ac)                   # should be 1

Problem 3 — Inclusion–exclusion and Boole’s inequality (14 pts)

Ross Chapter 2 references (read before solving):

  • §2.4 Some Simple Propositions (union formula; inclusion–exclusion; Boole’s inequality) 【Ch.2】 Strategy hints:
  • Derive P(E∪F) via Venn regions or by algebra: P(E)+P(F)−P(EF).
  • For three sets, write all seven regions and add/subtract carefully to get inclusion–exclusion.
  • Use Boole’s inequality as a quick upper bound; test tightness on your chosen example.

Starter R code:

# Starter: Inclusion–Exclusion & Boole's inequality
S <- 1:100  # toy finite space with equal probabilities
E <- which(S %% 2 == 0)   # even
F <- which(S %% 3 == 0)   # multiples of 3
G <- which(S %% 5 == 0)   # multiples of 5

P <- function(A) length(A)/length(S)
lhs2 <- P(union(E,F))
rhs2 <- P(E) + P(F) - P(intersect(E,F))
c(lhs2=lhs2, rhs2=rhs2, equal=abs(lhs2-rhs2) < 1e-12)

# Three-set inclusion–exclusion
EUF <- union(E,F)
lhs3 <- P(union(EUF, G))
rhs3 <- P(E) + P(F) + P(G) - P(intersect(E,F)) - P(intersect(E,G)) - P(intersect(F,G)) + P(Reduce(intersect, list(E,F,G)))
c(lhs3=lhs3, rhs3=rhs3, equal=abs(lhs3-rhs3) < 1e-12)

# Boole's inequality upper bound
upper <- P(E) + P(F) + P(G)
c(P_union=lhs3, upper_bound=upper)

Problem 5 — Matching (derangements) and “surprising” probabilities (12 pts)

Ross Chapter 2 references (read before solving):

  • §2.5 Equally Likely Outcomes — Matching Problem (Example 5m) 【Ch.2】
  • §2.5 Birthday problem (Example 5i) 【Ch.2】 Strategy hints:
  • For the matching (hats) problem, use inclusion–exclusion to obtain !n/n! = \sum_{k=0}^n (-1)^k/k!.
  • Note the limiting behavior !n/n! → e^{-1}; compare exact values to 1/e.
  • For 'surprising' cases (e.g., birthdays), compute the complement (no matches) and subtract from 1.

Starter R code:

# Starter: Matching (derangements) & surprising probabilities
# Exact derangement probability for N hats
derangement_prob <- function(N) {
  # P(no one gets own hat) = !N / N! = sum_{k=0}^N (-1)^k/k!
  sum(sapply(0:N, function(k) (-1)^k/factorial(k)))
}
sapply(c(5,10,20), derangement_prob)

# Approximation ~ 1/e
1/exp(1)

# Birthday problem: probability at least one match among n birthdays
birthday_at_least_one_match <- function(n, days=365) {
  1 - prod((days - 0:(n-1)) / days)
}
sapply(c(10,20,23,30), birthday_at_least_one_match)

# Starter: Applications – craps and blackjack

# --- Craps: exact probability of a win ---
# Compute from sum probabilities and conditional continuation
sum_prob <- table(rowSums(expand.grid(1:6,1:6))) / 36
p7  <- as.numeric(sum_prob["7"])
p11 <- as.numeric(sum_prob["11"])
p2  <- as.numeric(sum_prob["2"]); p3 <- as.numeric(sum_prob["3"]); p12 <- as.numeric(sum_prob["12"])
p_win_immediate <- p7 + p11
p_lose_immediate <- p2 + p3 + p12

# Points and their probabilities
pts <- c(4,5,6,8,9,10)
p_point <- as.numeric(sum_prob[as.character(pts)])
p_make_point <- c(3/36, 4/36, 5/36, 5/36, 4/36, 3/36)  # rolls equaling the point
p_seven      <- 6/36

p_win_from_point <- p_make_point / (p_make_point + p_seven)
p_win_total <- p_win_immediate + sum(p_point * p_win_from_point)
p_win_total  # ~ 0.49293

# --- Blackjack: probability that neither player nor dealer has blackjack (Monte Carlo starter) ---
deck <- expand.grid(value=c(2:10,"J","Q","K","A"), suit=c("S","H","D","C"))
card_value <- function(v) ifelse(v %in% c("J","Q","K"), 10, ifelse(v=="A", 11, as.numeric(v)))
is_blackjack <- function(cards) {
  vals <- card_value(cards$value)
  any(cards$value=="A") && sum(vals)==21
}
set.seed(631)
trials <- 5000
est <- mean(replicate(trials, {
  idx <- sample(1:nrow(deck), size=4, replace=FALSE)
  player <- deck[idx[1:2], ]; dealer <- deck[idx[3:4], ]
  !(is_blackjack(player) || is_blackjack(dealer))
}))
est
